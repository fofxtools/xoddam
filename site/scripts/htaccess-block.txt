# One canonical address, https://xoddam.com/: http -> https and www -> bare domain in a single 301.
# Certificate-validation paths are left alone so AutoSSL can renew the certificate.
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteCond %{HTTPS} off [OR]
  RewriteCond %{HTTP_HOST} ^www\. [NC]
  RewriteCond %{REQUEST_URI} !^/\.well-known/(acme-challenge|cpanel-dcv|pki-validation)/ [NC]
  RewriteRule ^ https://xoddam.com%{REQUEST_URI} [L,R=301]
</IfModule>

ErrorDocument 404 /404.html

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css text/plain text/xml application/javascript application/json application/xml image/svg+xml
</IfModule>

<IfModule mod_headers.c>
  # Built assets have a hash in their name, so they can be kept for good.
  <If "%{REQUEST_URI} =~ m#^/_astro/#">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </If>
  # Pictures keep their names: keep them for a week. (The game in /play/ sets its own rule, no-cache.)
  <FilesMatch "\.(webp|png|jpg|jpeg|woff2)$">
    Header set Cache-Control "public, max-age=604800"
  </FilesMatch>
</IfModule>
# --- end Xoddam ---
